    <script>
        const temas = [
            { id: 1, titulo: "Módulo 1: Introducción a la Lógica", desc: "Comprende qué es un algoritmo, un diagrama de flujo y cómo piensa una computadora." },
            { id: 2, titulo: "Módulo 2: Variables y Constantes", desc: "Aprende a almacenar datos temporales en memoria con diferentes tipos (texto, números)." },
            { id: 3, titulo: "Módulo 3: Estructuras Condicionales", desc: "Uso de operaciones lógicas y comandos condicionales como Si / If y Sino / Else." }
        ];

        document.addEventListener("DOMContentLoaded", renderMaterias);

        function renderMaterias() {
            let progreso = JSON.parse(localStorage.getItem("study_progreso_materias")) || {};
            let contenedor = document.getElementById("listaMaterias");
            contenedor.innerHTML = "";

            temas.forEach(t => {
                let completado = progreso[t.id] || false;
                contenedor.innerHTML += `
                    <div class="materia-item">
                        <div style="display:flex; justify-content:space-between; align-items:center;">
                            <strong>${t.titulo}</strong>
                            <span class="badge ${completado ? 'badge-done' : 'badge-pend'}">${completado ? 'Completado' : 'Pendiente'}</span>
                        </div>
                        <p style="color: #64748b; font-size: 0.9rem; margin: 0.5rem 0;">${t.desc}</p>
                        <button onclick="toggleTema(${t.id})" style="background-color: ${completado ? '#64748b' : '#10b981'}">
                            ${completado ? 'Marcar como Pendiente' : 'Marcar como Estudiado'}
                        </button>
                    </div>
                `;
            });
        }

        function toggleTema(id) {
            let progreso = JSON.parse(localStorage.getItem("study_progreso_materias")) || {};
            progreso[id] = !progreso[id];
            localStorage.setItem("study_progreso_materias", JSON.stringify(progreso));
            renderMaterias();
        }
    </script>
