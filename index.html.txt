    <script>
        // Inicialización y Carga de Datos desde el Navegador (LocalStorage)
        document.addEventListener("DOMContentLoaded", () => {
            cargarPerfil();
            cargarHorarios();
            cargarContenidos();
        });

        // 1. Manejo de Perfil
        function guardarPerfil() {
            const nombre = document.getElementById("nombreEstudiante").value;
            localStorage.setItem("study_nombre", nombre);
        }

        function cargarPerfil() {
            const nombreGuardado = localStorage.getItem("study_nombre");
            if (nombreGuardado) {
                document.getElementById("nombreEstudiante").value = nombreGuardado;
            }
        }

        // 2. Manejo de Horarios
        function agregarHorario() {
            const materia = document.getElementById("materia").value;
            const diaHora = document.getElementById("diaHora").value;

            if (!materia || !diaHora) {
                alert("Por favor completa ambos campos para registrar el horario.");
                return;
            }

            let horarios = JSON.parse(localStorage.getItem("study_horarios")) || [];
            horarios.push({ materia, diaHora });
            localStorage.setItem("study_horarios", JSON.stringify(horarios));

            document.getElementById("materia").value = "";
            document.getElementById("diaHora").value = "";
            cargarHorarios();
        }

        function cargarHorarios() {
            let horarios = JSON.parse(localStorage.getItem("study_horarios")) || [];
            let tbody = document.querySelector("#tablaHorarios tbody");
            tbody.innerHTML = "";

            horarios.forEach((h, index) => {
                let tr = document.createElement("tr");
                tr.innerHTML = `
                    <td>${h.materia}</td>
                    <td>${h.diaHora}</td>
                    <td><button style="background-color: #ef4444; padding: 0.3rem 0.6rem; font-size: 0.8rem;" onclick="eliminarHorario(${index})">Eliminar</button></td>
                `;
                tbody.appendChild(tr);
            });
        }

        function eliminarHorario(index) {
            let horarios = JSON.parse(localStorage.getItem("study_horarios")) || [];
            horarios.splice(index, 1);
            localStorage.setItem("study_horarios", JSON.stringify(horarios));
            cargarHorarios();
        }

        // 3. Contenidos de Programación y Progreso
        const contenidosBase = [
            { id: 1, titulo: "Introducción a la Lógica", desc: "¿Qué es un algoritmo y cómo se estructura?" },
            { id: 2, titulo: "Variables y Tipos de Datos", desc: "Cómo almacenar información en la memoria." },
            { id: 3, titulo: "Estructuras de Control (If / Else)", desc: "Tomar decisiones lógicas en el código." }
        ];

        function cargarContenidos() {
            let progreso = JSON.parse(localStorage.getItem("study_progreso")) || {};
            let contenedor = document.getElementById("listaContenidos");
            contenedor.innerHTML = "";

            contenidosBase.forEach(c => {
                let completado = progreso[c.id] || false;
                let div = document.createElement("div");
                div.style.marginBottom = "1rem";
                div.style.padding = "0.75rem";
                div.style.border = "1px solid #e2e8f0";
                div.style.borderRadius = "6px";
                div.innerHTML = `
                    <strong>${c.titulo}</strong> <span class="badge ${completado ? 'badge-done' : 'badge-pending'}">${completado ? 'Completado' : 'Pendiente'}</span>
                    <p style="margin: 0.3rem 0; font-size: 0.85rem; color: #64748b;">${c.desc}</p>
                    <button style="background-color: ${completado ? '#64748b' : '#10b981'}; font-size: 0.8rem; padding: 0.4rem 0.8rem; margin-top: 0.3rem;" onclick="toggleProgreso(${c.id})">
                        ${completado ? 'Marcar como Pendiente' : 'Marcar como Concluido'}
                    </button>
                `;
                contenedor.appendChild(div);
            });
        }

        function toggleProgreso(id) {
            let progreso = JSON.parse(localStorage.getItem("study_progreso")) || {};
            progreso[id] = !progreso[id];
            localStorage.setItem("study_progreso", JSON.stringify(progreso));
            cargarContenidos();
        }
    </script>
<script>
function mostrarPagina(pagina) {
  const secciones = {
    inicio: 'Perfil del Estudiante',
    horarios: 'Organización de Horarios',
    materias: 'Contenidos y progreso',
    ejercicios: 'Contenidos y Ejercicios Básicos de Programación',
    progreso: 'Contenidos y Progreso'
  };

  const destino = secciones[pagina];

  if (destino) {
    const elemento = [...document.querySelectorAll('h1,h2,h3,h4')]
      .find(e => e.textContent.trim().includes(destino));

    if (elemento) {
      elemento.scrollIntoView({ behavior: 'smooth' });
    }
  }
}
</script>
